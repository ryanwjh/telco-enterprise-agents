# Extra runtime dependencies for this agent's `tools/bigquery_ca.py`, which imports
# `google.adk.integrations.bigquery.BigQueryToolset`. That ADK module hard-imports
# `google.cloud.bigquery` and `google.cloud.dataplex_v1`, but neither ships with the
# base `google-adk` install — they're only pulled in by ADK's own `gcp`/`all`/`test`
# extras, none of which `adk deploy agent_engine` installs by default (it only
# installs `google-adk[a2a]`, see `_ensure_agent_engine_dependency` in
# `google/adk/cli/cli_deploy.py`).
#
# Without this file, `adk deploy agent_engine` auto-generates a requirements.txt
# containing only `google-adk[a2a]==<version>`, and the deployed container fails
# every request with `ImportError: cannot import name 'dataplex_v1' from
# 'google.cloud'` while loading root_agent.yaml (discovered during Assortment
# Planning's Task 12 post-deployment smoke check — every future agent scaffolded
# from this template would hit the identical failure on first deploy without this
# file). Versions below match this repo's pyproject.toml.
#
# `adk deploy agent_engine` appends `google-cloud-aiplatform[adk,agent_engines]`
# and `google-adk[a2a]==<version>` to this file automatically at deploy time.
google-cloud-bigquery>=3.0.0
google-cloud-dataplex>=1.0.0
